    <script>
        // ⚠️ 여기에 구글 앱스 스크립트로 배포한 웹앱 URL을 넣으세요!
        const SCRIPT_URL = "_구글_앱스_스크립트_URL을_넣으세요https://script.google.com/macros/s/AKfycbxv0g3o-blbEL-p1kZzbo7Hm_EIdXiy4ovRlh5KuXf9nRRXgL3ryP7sstKzIXP9-t_boQ/exec"; 
        
        let allItems = [];

        // 페이지 로드 시 구글 시트 데이터 가져오기
        document.addEventListener("DOMContentLoaded", () => {
            fetch(SCRIPT_URL)
                .then(response => response.json())
                .then(data => {
                    // 구글 시트에서 가져온 데이터(배열)를 저장
                    // 앱스 스크립트에서 { data: [...] } 형태로 넘긴다고 가정 (기존 기출문제 앱스스크립트 구조 참고)
                    allItems = data.data || data; 
                    
                    document.getElementById('loadingBar').style.display = 'none';
                    renderTable(allItems);
                })
                .catch(error => {
                    console.error('Error fetching data:', error);
                    document.getElementById('loadingBar').innerText = "❌ 데이터를 불러오지 못했습니다. URL을 확인해주세요.";
                    document.getElementById('loadingBar').style.backgroundColor = "#f8d7da";
                    document.getElementById('loadingBar').style.color = "#721c24";
                });
        });

        // 테이블 렌더링 함수
        function renderTable(data) {
            const tbody = document.getElementById('resultBody');
            const statBar = document.getElementById('statBar');
            tbody.innerHTML = '';
            
            statBar.innerText = `검색 결과: ${data.length}건`;

            if (data.length === 0) {
                tbody.innerHTML = '<tr><td colspan="17" style="text-align: center; padding: 30px; color: #666;">조건에 맞는 비품 데이터가 없습니다.</td></tr>';
                return;
            }

            // 17개 열을 모두 매핑하여 렌더링
            data.forEach(item => {
                // 키 이름은 구글 시트의 첫 번째 행(헤더)과 정확히 일치해야 합니다.
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td><span class="highlight">${item['장소명'] || ''}</span></td>
                    <td>${item['장소코드'] || ''}</td>
                    <td>${item['비품구분명'] || ''}</td>
                    <td>${item['비품구분코드'] || ''}</td>
                    <td>${item['비품유형명'] || ''}</td>
                    <td>${item['비품유형코드'] || ''}</td>
                    <td>${item['단가'] || ''}</td>
                    <td style="text-align: center; font-weight: bold;">${item['수량'] || ''}</td>
                    <td>${item['가격'] || ''}</td>
                    <td>${item['구매일'] || ''}</td>
                    <td style="font-weight: bold; color: #333;">${item['비품명'] || ''}</td>
                    <td>${item['구매자'] || ''}</td>
                    <td>${item['전표번호'] || ''}</td>
                    <td>${item['전표일자'] || ''}</td>
                    <td>${item['폐기일'] || ''}</td>
                    <td>${item['폐기담당자'] || ''}</td>
                    <td style="font-size: 13px; color: #555;">${item['비고'] || ''}</td>
                `;
                tbody.appendChild(tr);
            });
        }

        // 검색 및 필터링 함수
        function searchItems() {
            const query = document.getElementById('searchInput').value.toLowerCase().trim();
            
            if (!query) {
                renderTable(allItems);
                return;
            }
            
            const filtered = allItems.filter(item => {
                // 모든 열의 데이터를 하나의 문자열로 합쳐서 검색어가 포함되어 있는지 확인
                const rowString = Object.values(item).map(val => String(val).toLowerCase()).join(' ');
                return rowString.includes(query);
            });
            
            renderTable(filtered);
        }
    </script>
